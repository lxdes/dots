* Fedora
** Minimal TTY Wi-Fi Bootstrap (ThinkPad / Intel AX211)
#+BEGIN_SRC bash
# Connect via USB phone tethering or Ethernet adapter first, then:
sudo dnf install -y linux-firmware iwl*-firmware NetworkManager-wifi wpa_supplicant
sudo systemctl restart NetworkManager
nmcli device wifi connect "YOUR_SSID" password "YOUR_PASSWORD"
#+END_SRC

* Personal
** Setup Natural Scrolling + Tapping
#+BEGIN_SRC bash
sudo tee /etc/X11/xorg.conf.d/30-touchpad.conf > /dev/null <<'EOF'
Section "InputClass"
    Identifier "touchpad"
    Driver "libinput"
    MatchIsTouchpad "on"
    Option "Tapping" "on"
    Option "NaturalScrolling" "true"
EndSection
EOF
#+END_SRC

** Fix Track point 
#+BEGIN_SRC bash
sudo tee /etc/systemd/system/disable-trackpoint.service << 'EOF'
[Unit]
Description=Disable TrackPoint sensitivity, keep buttons
After=multi-user.target

[Service]
Type=oneshot
ExecStart=/bin/sh -c 'echo 0 > /sys/devices/platform/i8042/serio1/sensitivity'

[Install]
WantedBy=multi-user.target
EOF
sudo systemctl enable --now disable-trackpoint.service
#+END_SRC
